"use client";

import dynamic from "next/dynamic";
import Link from "next/link";
import { useState, useEffect } from "react";
import { motion } from "framer-motion";
import { Flame, Tag, ArrowRight, ShoppingBag } from "lucide-react";

// ── Context / Services
import { useRootData } from "@/src/context/rootContext";
import { getSlider } from "@/src/services/homePageService";

// ── Hooks

import { useDealsList, todayDealsFilters } from "@/src/hooks/useDealsList";

// ── Section Components (static imports)
import StatsSection from "@/src/components/Home/StatsSection";
import DealsSection from "@/src/components/Home/DealsSection";
import PromoSlider from "@/src/components/Home/PromoSlider";
import PlatformSection from "@/src/components/Home/PlatformSection";
import SectionHeader from "@/src/components/Common/SectionHeader";
import CommentModal from "@/src/components/deals/CommentModal";
import { fetchBulkCardComments } from "@/src/services/commentService";

// ── Lazy / no SSR — components that touch window, WebGL, or browser-only APIs
import HeroSection from "../components/Home/HeroSection";

const TelegramPopup = dynamic(
  () => import("../components/Home/TelegramPopup"),
  { ssr: false },
);

// ── Constants
const BL = "#0167ff";
const GR = "#34c759";

/* ─────────────────────────────────────────────
   HOME PAGE
───────────────────────────────────────────── */
export default function HomePage() {
  const [query, setQuery] = useState("");
  const [slides, setSlides] = useState([]);
  const { triggerSearch } = useRootData();

  // ── Reaction/Comment Data states
  const [commentPreviews, setCommentPreviews] = useState({});
  const [commentCounts, setCommentCounts] = useState({});
  const [reactionData, setReactionData] = useState({
    totals: {},
    userLiked: {},
  });
  const [commentModalDeal, setCommentModalDeal] = useState(null);

  // ── Today's Deals (API)
  const {
    deals: todayDeals,
    loading: todayLoading,
    reload: reloadTodayDeals,
  } = useDealsList({
    initialFilters: todayDealsFilters,
    autoLoad: false,
    limit: 8,
  });

  // ── Fetch both on mount
  useEffect(() => {
    reloadTodayDeals();
    getSlider()
      .then((d) => {
        if (d?.length) setSlides(d);
      })
      .catch(() => {});
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  // ── Fetch reactions in bulk
  useEffect(() => {
    if (!todayDeals.length) return;
    const dealIds = todayDeals.map((d) => d.id);
    const unloaded = dealIds.filter(
      (id) => reactionData.totals[id] === undefined,
    );
    if (!unloaded.length) return;

    import("@/src/services/reactionService")
      .then(({ getBulkReactions }) => getBulkReactions(unloaded))
      .then((data) => {
        if (data) {
          setReactionData((prev) => ({
            totals: { ...prev.totals, ...(data.totals || {}) },
            userLiked: { ...prev.userLiked, ...(data.userLiked || {}) },
          }));
        }
      })
      .catch(console.error);
  }, [todayDeals]);

  // ── Fetch comment previews in bulk
  useEffect(() => {
    if (!todayDeals.length) return;
    const dealIds = todayDeals.map((d) => d.id);
    const newIds = dealIds.filter((id) => commentPreviews[id] === undefined);
    if (!newIds.length) return;

    fetchBulkCardComments(newIds)
      .then((data) => {
        if (data?.previews)
          setCommentPreviews((prev) => ({ ...prev, ...data.previews }));
        if (data?.counts)
          setCommentCounts((prev) => ({ ...prev, ...data.counts }));
      })
      .catch(console.error);
  }, [todayDeals]);

  const handleSearch = (e) => {
    e.preventDefault();
    triggerSearch(query, true);
  };
  const handleChip = (q) => {
    setQuery(q);
    triggerSearch(q, true);
  };

  return (
    <main className="min-h-screen bg-[#f4f8ff]">
      <TelegramPopup />

      {/* ══ HERO ══════════════════════════════════════════ */}
      <HeroSection
        query={query}
        setQuery={setQuery}
        onSearch={handleSearch}
        onChip={handleChip}
      />

      {/* ══ BELOW-HERO SECTIONS ═══════════════════════════ */}
      <div className="mx-auto xl:container px-4 sm:px-6 lg:px-8 py-10 sm:py-14 space-y-10">
        {/* Divider */}
        <div
          className="h-px w-full"
          style={{
            background: `linear-gradient(to right,transparent,${BL}40,${GR}40,transparent)`,
          }}
        />

        {/* Stats */}
        <StatsSection />

        {/* ── Today's Best Deals (live API) */}
        <DealsSection
          deals={todayDeals}
          loading={todayLoading}
          title="Best Deals Today"
          label="Hot right now"
          icon={Flame}
          cta="View all deals"
          href="/best-deals"
          commentPreviews={commentPreviews}
          commentCounts={commentCounts}
          reactionData={reactionData}
          onViewComments={(d) => setCommentModalDeal(d)}
        />

        {/* Browse-all CTA */}
        <div className="flex justify-center -mt-8">
          <Link href="/best-deals">
            <motion.div
              whileHover={{ scale: 1.04 }}
              whileTap={{ scale: 0.97 }}
              data-cursor-label="Browse Deals"
              className="relative inline-flex items-center gap-2 px-7 py-3 rounded-xl font-black text-sm text-white overflow-hidden shadow-xl"
              style={{ background: `linear-gradient(135deg,${BL},${GR})` }}
            >
              <motion.div
                className="absolute inset-0 bg-gradient-to-r from-transparent via-white/15 to-transparent -skew-x-12"
                animate={{ x: ["-120%", "200%"] }}
                transition={{
                  duration: 3,
                  repeat: Infinity,
                  ease: "easeInOut",
                  repeatDelay: 1.2,
                }}
              />
              <ShoppingBag className="w-4 h-4 relative z-10" />
              <span className="relative z-10">Browse All Deals</span>
              <ArrowRight className="w-4 h-4 relative z-10" />
            </motion.div>
          </Link>
        </div>

        {/* Divider */}
        <div
          className="h-px w-full"
          style={{
            background: `linear-gradient(to right,transparent,${BL}40,${GR}40,transparent)`,
          }}
        />

        {/* Promo Slider */}
        {slides.length > 0 && (
          <section>
            <SectionHeader
              Icon={Tag}
              label="Exclusive offers"
              title="Featured Promotions"
            />
            <PromoSlider slides={slides} />
          </section>
        )}
        {/* Platform CTA */}
        <PlatformSection />
      </div>

      {/* Comment Modal */}
      <CommentModal
        deal={commentModalDeal}
        onClose={() => setCommentModalDeal(null)}
        onCommentCreated={(newCm) => {
          setCommentCounts((prev) => ({
            ...prev,
            [commentModalDeal.id]: (prev[commentModalDeal.id] || 0) + 1,
          }));
          setCommentPreviews((prev) => ({
            ...prev,
            [commentModalDeal.id]: [
              newCm,
              ...(prev[commentModalDeal.id] || []),
            ].slice(0, 2),
          }));
        }}
      />
    </main>
  );
}
