"use client";
import { compareProductUrl, searchProducts } from "@/src/api/pythonApiService";
import {
  AILoader,
  BL,
  EmptyState,
  InlineSearch,
  PlatformSection,
  ResultsBar,
} from "@/src/components/search/SearchComponents";
import { useRootData } from "@/src/context/rootContext";
import { AnimatePresence } from "framer-motion";
import { ArrowLeft } from "lucide-react";
import { useRouter, useSearchParams } from "next/navigation";
import { useCallback, useEffect, useMemo, useRef, useState } from "react";

export default function SearchPage() {
  const searchParams = useSearchParams();
  const router = useRouter();
  const query = searchParams.get("q");
  const urlParam = searchParams.get("url");

  const { isAuthenticated, isLoadingUser, setLoginModalOpen, triggerSearch } =
    useRootData();

  const [loading, setLoading] = useState(true);
  const [results, setResults] = useState(null);
  const [elapsed, setElapsed] = useState(0);
  const [activePlatform, setActivePlatform] = useState("All");

  const startRef = useRef(Date.now());
  const timerRef = useRef(null);
  const abortControllerRef = useRef(null);

  const startTimer = () => {
    startRef.current = Date.now();
    timerRef.current = setInterval(() => {
      setElapsed(Math.floor((Date.now() - startRef.current) / 1000));
    }, 1000);
  };

  const stopTimer = () => {
    clearInterval(timerRef.current);
    setElapsed(Math.floor((Date.now() - startRef.current) / 1000));
  };
  const performSearch = useCallback(
    async (q = query, u = urlParam) => {
      if (abortControllerRef.current) abortControllerRef.current.abort();
      const controller = new AbortController();
      abortControllerRef.current = controller;

      setLoading(true);
      setResults(null);
      setElapsed(0);
      startTimer();
      try {
        let data;
        if (u) data = await compareProductUrl(u, { signal: controller.signal });
        else if (q)
          data = await searchProducts(q, { signal: controller.signal });
        setResults(data);
      } catch (err) {
        if (
          ["CanceledError", "AbortError"].includes(err.name) ||
          err.message === "canceled"
        ) {
          console.log("Search request aborted");
          return;
        }
        console.error("Search failed:", err);
        setResults(null);
      } finally {
        if (abortControllerRef.current === controller) {
          stopTimer();
          setLoading(false);
          abortControllerRef.current = null;
        }
      }
    },
    [query, urlParam],
  );


  // ── Authentication Check Logic ──
  // useEffect(() => {
  //   // 1. Wait for user profile to finish loading
  //   if (isLoadingUser) return;

  //   // 2. Handle empty queries
  //   if (!query && !urlParam) {
  //     setLoading(false);
  //     return;
  //   }

  //   // 3. Only trigger login modal if strictly NOT authenticated
  //   if (!isAuthenticated) {
  //     setLoading(false);
  //     setLoginModalOpen(true);
  //     return;
  //   }

  //   // 4. Start search if authenticated
  //   performSearch();
  // }, [
  //   performSearch,
  //   isAuthenticated,
  //   isLoadingUser,
  //   query,
  //   urlParam,
  //   setLoginModalOpen,
  // ]);

  // ── Navigation Protection & Unmount Cleanup ──
  useEffect(() => {
    if (!loading) return;

    const handleBeforeUnload = (e) => {
      e.preventDefault();
      e.returnValue = "";
    };

    const handleClick = (e) => {
      const anchor = e.target.closest("a");
      if (anchor && anchor.href && !anchor.target) {
        try {
          const url = new URL(anchor.href, window.location.href);
          if (
            url.pathname !== window.location.pathname ||
            url.search !== window.location.search
          ) {
            if (
              !window.confirm(
                "A search is currently in progress. Are you sure you want to leave this page?",
              )
            ) {
              e.preventDefault();
              e.stopPropagation();
            }
          }
        } catch (err) {}
      }
    };

    window.addEventListener("beforeunload", handleBeforeUnload);
    document.addEventListener("click", handleClick, { capture: true });

    return () => {
      window.removeEventListener("beforeunload", handleBeforeUnload);
      document.removeEventListener("click", handleClick, { capture: true });
    };
  }, [loading]);

  useEffect(() => {
    return () => {
      clearInterval(timerRef.current);
      if (abortControllerRef.current) abortControllerRef.current.abort();
    };
  }, []);

  const handleInlineSearch = (newQ) => triggerSearch(newQ);

  const allProducts = useMemo(() => {
    if (!results?.platforms) return [];
    return Object.values(results.platforms).flat();
  }, [results]);

  const platformList = useMemo(() => {
    if (!results?.platforms) return ["All"];
    return [
      "All",
      ...Object.keys(results.platforms).filter(
        (p) => results.platforms[p]?.length > 0,
      ),
    ];
  }, [results]);

  const filteredProducts = useMemo(() => {
    if (activePlatform === "All") return allProducts;
    return results?.platforms[activePlatform] || [];
  }, [activePlatform, allProducts, results]);

  return (
    <div className="min-h-screen bg-[#f8f9fb] pb-20">
      <AnimatePresence>
        {loading && <AILoader query={query || urlParam} elapsed={elapsed} />}
      </AnimatePresence>

      <header className="sticky top-0 z-40 bg-white/80 backdrop-blur-md border-b border-gray-100 px-4 py-3">
        <div className="max-w-7xl mx-auto flex flex-col gap-1">
          <div className="flex items-center gap-3">
            <button
              onClick={() => router.push("/")}
              className="p-2 hover:bg-gray-100 rounded-full transition-colors"
            >
              <ArrowLeft className="w-5 h-5 text-gray-600" />
            </button>
            <div className="flex-1 flex flex-col min-w-0">
              <span className="text-[10px] font-black uppercase tracking-widest text-gray-400">
                AI Search Active
              </span>
              <h1 className="text-sm font-black text-gray-900 truncate">
                {query || "Comparing URL..."}
              </h1>
            </div>
          </div>
          <InlineSearch currentQuery={query} onSearch={handleInlineSearch} />
        </div>
      </header>

      {!loading && results && (
        <main className="max-w-7xl mx-auto">
          <ResultsBar count={allProducts.length} elapsed={elapsed} />

          <div className="flex items-center gap-2 overflow-x-auto px-4 py-4 no-scrollbar">
            {platformList.map((p) => (
              <button
                key={p}
                onClick={() => setActivePlatform(p)}
                className="px-4 py-2 rounded-2xl text-xs font-black whitespace-nowrap transition-all border"
                style={{
                  background: activePlatform === p ? BL : "white",
                  color: activePlatform === p ? "white" : "#64748b",
                  borderColor: activePlatform === p ? BL : "#e2e8f0",
                  boxShadow:
                    activePlatform === p ? `0 4px 12px ${BL}30` : "none",
                }}
              >
                {p}
              </button>
            ))}
          </div>

          <div className="px-4">
            {activePlatform === "All" ? (
              Object.keys(results.platforms || {}).map((p) => (
                <PlatformSection
                  key={p}
                  platform={p}
                  products={results.platforms[p]}
                />
              ))
            ) : (
              <PlatformSection
                platform={activePlatform}
                products={filteredProducts}
              />
            )}
          </div>
        </main>
      )}

      {!loading && results && allProducts.length === 0 && (
        <EmptyState
          query={query}
          onBack={() => router.push("/")}
          elapsed={elapsed}
        />
      )}
    </div>
  );
}
