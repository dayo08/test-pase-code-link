import React from "react";
import { FiAlertCircle, FiTrash2, FiX, FiLoader } from "react-icons/fi";
import Modal from "../Common/Modal";

export default function DeleteAlbum({
  state,
  updateState,
  confirmDelete,
  isLoading,
  albumName = "this album",
}) {
  const handleClose = () => {
    if (!isLoading) {
      updateState({ showDeleteModal: false });
    }
  };

  const handleConfirmDelete = () => {
    if (!isLoading) {
      confirmDelete();
    }
  };

  return (
    <Modal
      isOpen={state}
      onClose={handleClose}
      closeOnOutsideClick={false}
      className="bg-white rounded-lg shadow-xl max-w-md w-full mx-4 overflow-hidden"
    >
      {/* Header */}
      <div className="flex items-center justify-between p-6 border-b border-gray-200">
        <div className="flex items-center space-x-3">
          <div className="flex-shrink-0 w-10 h-10 bg-red-100 rounded-full flex items-center justify-center">
            {isLoading ? (
              <FiLoader className="w-5 h-5 text-red-600 animate-spin" />
            ) : (
              <FiAlertCircle className="w-5 h-5 text-red-600" />
            )}
          </div>
          <h3 className="text-lg font-semibold text-gray-900">
            {isLoading ? "Deleting Album..." : "Delete Album"}
          </h3>
        </div>
        {!isLoading && (
          <button
            onClick={handleClose}
            className="text-gray-400 hover:text-gray-600 transition-colors p-1"
            aria-label="Close modal"
          >
            <FiX className="w-5 h-5" />
          </button>
        )}
      </div>

      {/* Content */}
      <div className="p-6">
        <p className="text-gray-600 mb-4">
          {isLoading
            ? "Please wait while we delete your album. This action cannot be undone."
            : `Are you sure you want to permanently delete "${albumName}"? This action cannot be undone and all photos in this album will be removed.`}
        </p>

        {isLoading && (
          <div className="bg-gray-50 rounded-lg p-4 mb-4">
            <div className="flex items-center space-x-3">
              <FiLoader className="w-4 h-4 text-gray-600 animate-spin" />
              <span className="text-sm text-gray-600">
                Deleting album and all associated data...
              </span>
            </div>
          </div>
        )}
      </div>

      {/* Footer */}
      <div className="px-6 py-4 bg-gray-50 border-t border-gray-200">
        {isLoading ? (
          <div className="flex items-center justify-center space-x-2 text-gray-600">
            <FiLoader className="w-4 h-4 animate-spin" />
            <span className="text-sm">Processing deletion...</span>
          </div>
        ) : (
          <div className="flex items-center justify-end space-x-3">
            <button
              onClick={handleClose}
              className="px-4 py-2 text-gray-700 bg-white border border-gray-300 rounded-md hover:bg-gray-50 focus:outline-none focus:ring-2 focus:ring-offset-2 focus:ring-gray-500 transition-colors"
              disabled={isLoading}
            >
              Cancel
            </button>
            <button
              onClick={handleConfirmDelete}
              className="px-4 py-2 bg-red-600 text-white rounded-md hover:bg-red-700 focus:outline-none focus:ring-2 focus:ring-offset-2 focus:ring-red-500 transition-colors flex items-center space-x-2"
              disabled={isLoading}
            >
              <FiTrash2 className="w-4 h-4" />
              <span>Delete Album</span>
            </button>
          </div>
        )}
      </div>
    </Modal>
  );
}
