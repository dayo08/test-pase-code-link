// src/components/EventAlbum/UploadManager.jsx
"use client";
import React, { useMemo, useState } from "react";
import {
  FiCheckCircle,
  FiUpload,
  FiX,
  FiClock,
  FiZap,
  FiWifi,
  FiPackage,
  FiActivity,
  FiChevronDown,
  FiChevronUp,
  FiAlertCircle,
  FiLoader,
} from "react-icons/fi";

/* ─── SUB-COMPONENTS ────────────────────────────────────────────── */

const HeroStat = ({
  icon: Icon,
  label,
  value,
  subValue,
  highlight = false,
  theme,
}) => (
  <div
    className={`p-2.5 rounded-xl border transition-all duration-200 ${
      highlight
        ? "bg-white/10 border-white/20 shadow-md shadow-black/10"
        : "bg-black/10 border-white/5 active:bg-black/20"
    }`}
  >
    <div className="flex items-center gap-1.5 mb-1.5">
      <div
        className="p-1 rounded-md"
        style={{
          backgroundColor: highlight
            ? theme.secondary
            : "rgba(255,255,255,0.1)",
          color: highlight ? theme.primary : theme.secondary,
        }}
      >
        <Icon className="w-3 h-3" />
      </div>
      <span className="text-[9px] font-black uppercase tracking-widest text-white/50">
        {label}
      </span>
    </div>
    <div className="flex items-baseline gap-1">
      <span className="text-base font-black tracking-tight text-white leading-none tabular-nums">
        {value}
      </span>
      {subValue && (
        <span className="text-[9px] font-bold text-white/40">{subValue}</span>
      )}
    </div>
  </div>
);

const FileItem = ({ file, theme }) => {
  const isFailed = file.status === "failed";
  const isDone = file.status === "completed";
  const isProcessing = file.status === "processing";
  const isUploading = file.status === "uploading";

  return (
    <div
      className="group bg-white border border-gray-100 rounded-xl p-2 transition-all duration-150 active:scale-[0.99]"
      style={{
        borderColor: isDone
          ? "rgba(34,197,94,0.15)"
          : isFailed
            ? "rgba(239,68,68,0.15)"
            : undefined,
      }}
    >
      <div className="flex items-center justify-between gap-2">
        <div className="flex items-center gap-2 min-w-0 flex-1">
          <div
            className={`w-7 h-7 rounded-lg flex items-center justify-center shrink-0 transition-all ${
              isFailed
                ? "bg-red-50 text-red-500"
                : isDone
                  ? "bg-green-50 text-green-600"
                  : "bg-gray-50"
            }`}
            style={!isFailed && !isDone ? { color: theme.primary } : {}}
          >
            {isDone ? (
              <FiCheckCircle className="w-3.5 h-3.5" />
            ) : isProcessing ? (
              <FiZap
                className="w-3.5 h-3.5 animate-pulse"
                style={{ color: theme.secondary }}
              />
            ) : isUploading ? (
              <FiLoader
                className="w-3.5 h-3.5 animate-spin"
                style={{ color: theme.third }}
              />
            ) : isFailed ? (
              <FiAlertCircle className="w-3.5 h-3.5" />
            ) : (
              <FiPackage className="w-3.5 h-3.5" />
            )}
          </div>
          <div className="min-w-0">
            <p
              className="text-[11px] font-semibold truncate leading-tight"
              style={{ color: theme.primary }}
            >
              {file.originalName}
            </p>
            <p className="text-[9px] font-black uppercase tracking-wider text-gray-400 mt-px">
              {isFailed
                ? "Failed"
                : isProcessing
                  ? "Optimizing…"
                  : isUploading
                    ? "Uploading…"
                    : isDone
                      ? "Done"
                      : "Waiting"}
            </p>
          </div>
        </div>
        {(isUploading || isProcessing) && (
          <span
            className="text-[10px] font-black tabular-nums shrink-0"
            style={{ color: theme.primary }}
          >
            {file.progress || 0}%
          </span>
        )}
      </div>
      {(isUploading || isProcessing) && (
        <div className="mt-2 h-0.5 w-full bg-gray-100 rounded-full overflow-hidden">
          <div
            className="h-full rounded-full transition-all duration-300"
            style={{
              width: `${file.progress || 0}%`,
              background: isProcessing ? theme.gradient : theme.primary,
            }}
          />
        </div>
      )}
    </div>
  );
};

/* ─── MAIN COMPONENT ───────────────────────────────────────────── */

const UploadManager = ({ uploadTracker, onCancel, onClose }) => {
  const {
    files,
    isUploading,
    totalProgress,
    totalFiles,
    results,
    showResults,
    networkSpeed,
    speedQuality,
    speedColor,
    estimatedTimeRemaining,
  } = uploadTracker;

  const theme = {
    primary: "#343148",
    secondary: "#d7c39d",
    third: "#d7c39d",
    gradient: "linear-gradient(135deg, #343148 100%)",
  };

  const [expanded, setExpanded] = useState(false);

  const stats = useMemo(() => {
    const s = {
      completed: results.successCount || 0,
      failed: results.failedCount || 0,
      processing: files.filter((f) => f.status === "processing").length,
      uploading: files.filter((f) => f.status === "uploading").length,
      pending: files.filter(
        (f) => f.status === "pending" || f.status === "ready",
      ).length,
    };
    s.totalMB = files.reduce((acc, f) => acc + (f.size || 0), 0).toFixed(1);
    return s;
  }, [files, results]);

  const formatTime = (seconds) => {
    if (!seconds || seconds <= 0) return "--";
    const mins = Math.floor(seconds / 60);
    const secs = seconds % 60;
    return mins > 0 ? `${mins}m ${secs}s` : `${secs}s`;
  };

  return (
    <>
      {/* Backdrop */}
      <div
        className="fixed inset-0 backdrop-blur-md z-[100] animate-in fade-in duration-300"
        style={{ backgroundColor: `${theme.primary}55` }}
        onClick={!isUploading ? onClose : undefined}
      />

      {/* Sheet */}
      <div className="fixed inset-x-0 bottom-0 sm:inset-0 sm:flex sm:items-center sm:justify-center z-[101] p-0 sm:p-4">
        <div className="w-full max-w-lg bg-white rounded-t-3xl sm:rounded-3xl shadow-2xl overflow-hidden flex flex-col animate-in slide-in-from-bottom duration-400 max-h-[93vh]">
          {/* ── DRAG HANDLE (mobile) ── */}
          <div className="flex justify-center pt-2.5 pb-1 sm:hidden">
            <div className="w-8 h-1 rounded-full bg-gray-200" />
          </div>

          {/* ── HERO ── */}
          <div
            className="pt-5 pb-6 px-4 relative"
            style={{ background: theme.gradient }}
          >
            {/* Close */}
            <button
              onClick={isUploading ? onCancel : onClose}
              className="absolute top-3 right-3 p-1.5 bg-white/10 hover:bg-white/20 active:scale-90 rounded-full text-white/60 hover:text-white transition-all duration-150"
            >
              <FiX className="w-4 h-4" />
            </button>

            {/* Icon + Title */}
            <div className="flex flex-col items-center text-center gap-3 mb-6">
              <div className="w-12 h-12 rounded-2xl flex items-center justify-center shadow-xl shadow-black/20 shrink-0 animate-in zoom-in duration-500 bg-white/20 backdrop-blur-sm">
                {isUploading ? (
                  <FiActivity className="w-6 h-6 text-white animate-pulse" />
                ) : (
                  <FiCheckCircle className="w-6 h-6 text-white" />
                )}
              </div>
              <div>
                <h1 className="text-lg font-black text-white leading-tight uppercase tracking-tight">
                  {isUploading ? "Syncing Media" : "Upload Complete"}
                </h1>
                <p className="text-[9px] font-black text-white/70 uppercase tracking-[0.25em] mt-0.5">
                  {isUploading
                    ? `Processing ${totalFiles} files`
                    : "All files secured"}
                </p>
              </div>
            </div>

            {/* Progress Bar */}
            <div className="mb-5 px-1">
              <div className="flex justify-between items-end mb-1.5">
                <span className="text-[9px] font-black uppercase tracking-widest text-white/50">
                  Overall Progress
                </span>
                <span className="text-2xl font-black text-white tabular-nums tracking-tighter leading-none">
                  {totalProgress}%
                </span>
              </div>
              <div className="h-2 w-full bg-white/10 rounded-full overflow-hidden">
                <div
                  className="h-full bg-white/80 rounded-full transition-all duration-700 relative overflow-hidden"
                  style={{ width: `${totalProgress}%` }}
                >
                  <div className="absolute inset-0 bg-white/30 animate-pulse rounded-full" />
                </div>
              </div>
            </div>

            {/* Stats Grid */}
            <div className="grid grid-cols-2 gap-2">
              <HeroStat
                icon={FiWifi}
                label={`Speed (${speedQuality || "..."})`}
                value={networkSpeed || "0.0"}
                subValue="Mbps"
                highlight={speedColor === "green"}
                theme={theme}
              />

              <HeroStat
                icon={FiClock}
                label="Time Left"
                value={formatTime(estimatedTimeRemaining)}
                theme={theme}
              />
              <HeroStat
                icon={FiZap}
                label="Processed"
                value={stats.totalMB}
                subValue="MB"
                theme={theme}
              />
              <HeroStat
                icon={FiPackage}
                label="Queue"
                value={stats.pending + stats.uploading + stats.processing}
                subValue={`/ ${totalFiles}`}
                theme={theme}
              />
            </div>
          </div>

          {/* ── FILE LIST ── */}
          <div className="flex-1 overflow-y-auto px-4 py-3 bg-gray-50/60">
            <div className="flex items-center justify-between mb-2.5">
              <span className="text-[9px] font-black text-gray-400 uppercase tracking-widest">
                Active Files
              </span>
              <button
                onClick={() => setExpanded(!expanded)}
                className="flex items-center gap-1 text-[9px] font-black uppercase tracking-widest transition-all active:scale-95"
                style={{ color: theme.primary }}
              >
                {expanded ? "Collapse" : "View All"}
                {expanded ? (
                  <FiChevronUp className="w-3 h-3" />
                ) : (
                  <FiChevronDown className="w-3 h-3" />
                )}
              </button>
            </div>

            <div className="space-y-1.5">
              {files
                .filter(
                  (f) =>
                    f.status === "uploading" ||
                    f.status === "processing" ||
                    (expanded && f.status === "pending"),
                )
                .slice(0, expanded ? 50 : 5)
                .map((file) => (
                  <FileItem key={file.hash} file={file} theme={theme} />
                ))}

              {files.length > 5 && !expanded && (
                <p className="text-center text-[9px] font-bold text-gray-400 tracking-wider py-1.5">
                  +{files.length - 5} more in queue
                </p>
              )}

              {files.length === 0 && !isUploading && !showResults && (
                <div className="py-10 text-center">
                  <div className="w-10 h-10 bg-green-50 text-green-500 rounded-2xl flex items-center justify-center mx-auto mb-2.5">
                    <FiCheckCircle className="w-5 h-5" />
                  </div>
                  <p
                    className="text-xs font-bold"
                    style={{ color: theme.primary }}
                  >
                    All files verified
                  </p>
                  <p className="text-[9px] text-gray-400 font-medium mt-0.5 uppercase tracking-widest">
                    Securely stored
                  </p>
                </div>
              )}

              {showResults && (
                <div className="py-6 px-2 bg-white rounded-2xl border border-gray-100 shadow-sm animate-in fade-in zoom-in duration-300">
                  <div className="flex items-center justify-around text-center">
                    <div className="space-y-1">
                      <p className="text-[9px] font-black text-gray-400 uppercase tracking-widest">
                        Successful
                      </p>
                      <p className="text-xl font-black text-green-500">
                        {results.successCount}
                      </p>
                    </div>
                    <div className="w-px h-8 bg-gray-100" />
                    <div className="space-y-1">
                      <p className="text-[9px] font-black text-gray-400 uppercase tracking-widest">
                        Failed
                      </p>
                      <p
                        className={`text-xl font-black ${results.failedCount > 0 ? "text-red-500" : "text-gray-300"}`}
                      >
                        {results.failedCount}
                      </p>
                    </div>
                    <div className="w-px h-8 bg-gray-100" />
                    <div className="space-y-1">
                      <p className="text-[9px] font-black text-gray-400 uppercase tracking-widest">
                        Total
                      </p>
                      <p
                        className="text-xl font-black"
                        style={{ color: theme.primary }}
                      >
                        {totalFiles}
                      </p>
                    </div>
                  </div>
                  {results.failedCount > 0 && (
                    <div className="mt-4 p-3 bg-red-50 rounded-xl border border-red-100">
                      <p className="text-[10px] font-bold text-red-600 flex items-center gap-1.5">
                        <FiAlertCircle className="w-3 h-3" />
                        Some files failed to upload. You can retry them later.
                      </p>
                    </div>
                  )}
                </div>
              )}
            </div>
          </div>

          {/* ── FOOTER ── */}
          <div className="px-4 py-4 bg-white border-t border-gray-100">
            <div className="flex items-center justify-between gap-3">
              <div className="min-w-0">
                <p
                  className="text-[9px] font-black uppercase tracking-widest mb-1"
                  style={{ color: theme.primary }}
                >
                  Status
                </p>
                <div className="flex items-center gap-1.5">
                  <div
                    className={`w-1.5 h-1.5 rounded-full shrink-0 ${isUploading ? "animate-pulse" : ""}`}
                    style={{
                      backgroundColor: isUploading ? theme.third : "#22c55e",
                    }}
                  />
                  <p className="text-[10px] font-semibold text-gray-500 truncate">
                    {isUploading
                      ? `Batch processing… (${stats.completed}/${totalFiles})`
                      : "Sync complete"}
                  </p>
                </div>
              </div>

              {isUploading ? (
                <button
                  onClick={onCancel}
                  className="px-5 py-2.5 bg-gray-100 hover:bg-gray-200 rounded-xl text-[10px] font-black uppercase tracking-widest transition-all active:scale-95 shrink-0"
                  style={{ color: theme.primary }}
                >
                  Stop
                </button>
              ) : (
                <button
                  onClick={onClose}
                  className="px-6 py-2.5 text-white rounded-xl text-[10px] font-black uppercase tracking-widest transition-all hover:opacity-90 active:scale-95 shadow-lg shrink-0"
                  style={{
                    background: theme.gradient,
                    boxShadow: `0 6px 20px ${theme.primary}33`,
                  }}
                >
                  Done
                </button>
              )}
            </div>
          </div>
        </div>
      </div>
    </>
  );
};

export default UploadManager;
