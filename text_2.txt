"use client";
import { useRef } from "react";
import { motion, useScroll, useTransform } from "framer-motion";
import { DealCard } from "@/src/components/deals/DealCard";
import SectionHeader from "@/src/components/Common/SectionHeader";

const BL = "#0167ff";

// Skeleton for loading state
export function DealSkeletonGrid({ count = 8 }) {
  return (
    <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 gap-3 sm:gap-4">
      {Array.from({ length: count }).map((_, i) => (
        <div
          key={i}
          className="rounded-2xl border border-gray-100 bg-white overflow-hidden animate-pulse shadow-sm"
        >
          <div className="aspect-[4/3] bg-gradient-to-br from-gray-100 to-gray-50" />
          <div className="p-3 space-y-2.5">
            <div className="h-3 bg-gray-100 rounded-full w-4/5" />
            <div className="h-3 bg-gray-100 rounded-full w-3/5" />
            <div className="h-px bg-gray-100 w-full" />
            <div className="h-4 bg-gray-100 rounded-full w-2/5" />
          </div>
        </div>
      ))}
    </div>
  );
}

export default function DealsSection({
  deals = [],
  loading = false,
  title,
  label,
  icon: Icon,
  cta,
  href,
  commentPreviews = {},
  commentCounts = {},
  reactionData = { totals: {}, userLiked: {} },
  onViewComments,
}) {
  const sectionRef = useRef(null);
  const { scrollYProgress } = useScroll({
    target: sectionRef,
    offset: ["start end", "end start"],
  });
  const bgY = useTransform(scrollYProgress, [0, 1], [-20, 20]);

  return (
    <section ref={sectionRef} className="relative">
      <motion.div
        className="absolute inset-0 pointer-events-none rounded-3xl opacity-30"
        style={{
          y: bgY,
          background: `radial-gradient(ellipse at 50% 50%,${BL}08,transparent 70%)`,
        }}
      />
      <SectionHeader
        Icon={Icon}
        label={label}
        title={title}
        cta={cta}
        href={href}
      />

      {loading ? (
        <DealSkeletonGrid count={8} />
      ) : deals.length === 0 ? (
        <p className="text-center text-sm text-gray-400 py-12">
          No deals available right now.
        </p>
      ) : (
        <motion.div 
          initial="hidden"
          whileInView="show"
          viewport={{ once: true, margin: "-10% 0px" }}
          variants={{
            show: {
              transition: {
                staggerChildren: 0.05,
              }
            }
          }}
          className="grid grid-cols-2 sm:grid-cols-3 xl:grid-cols-4 gap-2.5 md:gap-3.5"
        >
          {deals.map((deal, i) => (
            <motion.div 
              key={deal.id || i}
              variants={{
                hidden: { opacity: 0, y: 20 },
                show: { opacity: 1, y: 0 }
              }}
              transition={{ duration: 0.4, ease: "easeOut" }}
            >
              <DealCard 
                deal={deal} 
                index={i} 
                commentPreviews={commentPreviews[deal.id] || []}
                commentCount={commentCounts[deal.id] || 0}
                reactionTotal={reactionData.totals[deal.id] || 0}
                userLiked={reactionData.userLiked[deal.id] || false}
                onViewComments={onViewComments}
              />
            </motion.div>
          ))}
        </motion.div>
      )}
    </section>
  );
}
