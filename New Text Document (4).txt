import React from "react";
import { MdDeleteOutline, MdClose, MdImage, MdWarning } from "react-icons/md";
import { IoMdImages } from "react-icons/io";
import { BiLoader } from "react-icons/bi";
import Modal from "../Common/Modal";

const DeleteImage = ({ isOpen, onClose, onConfirm, count, isLoading }) => {
  if (!isOpen) return null;

  const handleClose = () => {
    if (!isLoading) {
      onClose();
    }
  };

  const handleConfirm = () => {
    if (!isLoading) {
      onConfirm();
    }
  };

  return (
    <Modal
      isOpen={isOpen}
      onClose={handleClose}
      className="bg-white rounded-2xl shadow-2xl max-w-md w-full mx-4 overflow-hidden"
      closeOnOutsideClick={!isLoading}
    >
      {/* Header */}
      <div className="relative px-6 pt-6 pb-4">
        {!isLoading && (
          <button
            onClick={handleClose}
            className="absolute top-4 right-4 p-2 hover:bg-gray-100 rounded-full transition-all duration-200 group"
            aria-label="Close modal"
          >
            <MdClose className="text-gray-400 group-hover:text-gray-600 text-xl" />
          </button>
        )}

        <div className="text-center">
          <div className="mx-auto w-16 h-16 bg-gradient-to-br from-red-100 to-red-50 rounded-full flex items-center justify-center mb-4 shadow-sm">
            {isLoading ? (
              <BiLoader className="text-red-600 text-2xl animate-spin" />
            ) : (
              <MdWarning className="text-red-600 text-2xl" />
            )}
          </div>

          <h3 className="text-xl font-bold text-gray-900 mb-2">
            {isLoading
              ? "Deleting Images..."
              : `Delete ${count > 1 ? `${count} Images` : "Image"}?`}
          </h3>

          <p className="text-gray-600 text-sm leading-relaxed">
            {isLoading
              ? "Please wait while we process your request. This may take a moment."
              : "This action cannot be undone. The selected images will be permanently removed."}
          </p>
        </div>
      </div>

      {/* Content */}
      <div className="px-6 pb-4">
        {!isLoading ? (
          <div className="bg-gradient-to-r from-gray-50 to-gray-100 rounded-xl p-4 border border-gray-200">
            <div className="flex items-center justify-center gap-3">
              <div className="flex-shrink-0 w-10 h-10 bg-white rounded-lg flex items-center justify-center shadow-sm">
                {count > 1 ? (
                  <IoMdImages className="text-gray-500 text-lg" />
                ) : (
                  <MdImage className="text-gray-500 text-lg" />
                )}
              </div>
              <div className="text-center">
                <p className="text-sm font-medium text-gray-700">
                  {count > 1 ? `${count} images selected` : "1 image selected"}
                </p>
                <p className="text-xs text-gray-500 mt-1">
                  {count > 1
                    ? "All selected images will be deleted"
                    : "This image will be deleted"}
                </p>
              </div>
            </div>
          </div>
        ) : (
          <div className="bg-gray-50 rounded-xl p-4 border border-gray-200">
            <div className="flex items-center justify-center gap-3 mb-3">
              <BiLoader className="text-red-500 text-lg animate-spin" />
              <span className="text-sm font-medium text-gray-700">
                Processing deletion...
              </span>
            </div>
            <div className="w-full bg-gray-200 rounded-full h-2 overflow-hidden">
              <div
                className="bg-gradient-to-r from-red-500 to-red-600 h-2 rounded-full animate-pulse"
                style={{ width: "75%" }}
              ></div>
            </div>
          </div>
        )}
      </div>

      {/* Actions */}
      <div className="px-6 pb-6">
        {!isLoading ? (
          <div className="space-y-3">
            <button
              onClick={handleConfirm}
              className="w-full py-3 bg-gradient-to-r from-red-600 to-red-700 text-white font-semibold rounded-xl hover:from-red-700 hover:to-red-800 transition-all duration-200 focus:outline-none focus:ring-2 focus:ring-red-500 focus:ring-offset-2 shadow-lg hover:shadow-xl transform hover:scale-[1.02] active:scale-[0.98]"
            >
              <span className="flex items-center justify-center gap-2">
                <MdDeleteOutline className="text-lg" />
                Delete {count > 1 ? "Images" : "Image"}
              </span>
            </button>

            <button
              onClick={handleClose}
              className="w-full py-3 border-2 border-gray-200 text-gray-700 font-semibold rounded-xl hover:bg-gray-50 hover:border-gray-300 transition-all duration-200 focus:outline-none focus:ring-2 focus:ring-gray-400 focus:ring-offset-2 transform hover:scale-[1.01] active:scale-[0.99]"
            >
              Cancel
            </button>
          </div>
        ) : (
          <div className="text-center">
            <div className="inline-flex items-center gap-2 px-4 py-2 bg-gray-100 rounded-full">
              <BiLoader className="text-gray-600 text-sm animate-spin" />
              <span className="text-sm text-gray-600 font-medium">
                Please wait...
              </span>
            </div>
          </div>
        )}
      </div>
    </Modal>
  );
};

export default DeleteImage;
