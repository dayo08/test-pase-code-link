import config from "@/src/config/config";
import { useAlbum } from "@/src/context/albumContext";
import dynamic from "next/dynamic";
import Image from "next/image";
import { useState } from "react";
import {
  FiCheck,
  FiChevronLeft,
  FiChevronRight,
  FiEye,
  FiImage,
  FiTrash2,
  FiX,
} from "react-icons/fi";
import { MdCheckBoxOutlineBlank, MdSelectAll } from "react-icons/md";

const DeleteImage = dynamic(() => import("../EventAlbum/DeleteImage"), {
  loading: () => <p>Loading...</p>,
  ssr: false,
});

const ImageGallery = ({
  uploadedImages,
  pagination,
  onPageChange,
  onLimitChange,
  currentAlbumId,
}) => {
  const { removeImages } = useAlbum();
  const [selectedImages, setSelectedImages] = useState(new Set());
  const [lastSelectedIndex, setLastSelectedIndex] = useState(null); // **NEW STATE**
  const [deleteModal, setDeleteModal] = useState({
    isOpen: false,
    imageIds: [],
  });
  const [isDeleting, setIsDeleting] = useState(false);
  const isSelectionMode = selectedImages.size > 0;

  // **ENHANCED: Handle single image selection with Shift+Click support**
  const toggleImageSelection = (imageId, index, event) => {
    // **SHIFT+CLICK: Range selection**
    if (event?.shiftKey && lastSelectedIndex !== null) {
      const newSelection = new Set(selectedImages);
      const [start, end] = [lastSelectedIndex, index].sort((a, b) => a - b);

      // Select all images in the range
      for (let i = start; i <= end; i++) {
        if (uploadedImages[i]) {
          newSelection.add(uploadedImages[i]._id);
        }
      }

      setSelectedImages(newSelection);
      setLastSelectedIndex(index);
    }
    // **REGULAR CLICK: Toggle single selection**
    else {
      const newSelection = new Set(selectedImages);
      if (newSelection.has(imageId)) {
        newSelection.delete(imageId);
      } else {
        newSelection.add(imageId);
      }
      setSelectedImages(newSelection);
      setLastSelectedIndex(index);
    }
  };

  // Select all images
  const selectAllImages = () => {
    const allImageIds = new Set(uploadedImages.map((img) => img._id));
    setSelectedImages(allImageIds);
  };

  // Deselect all images
  const deselectAllImages = () => {
    setSelectedImages(new Set());
    setLastSelectedIndex(null); // **RESET LAST SELECTED**
  };

  // Handle multi delete
  const handleMultiDelete = () => {
    if (selectedImages.size > 0) {
      setDeleteModal({
        isOpen: true,
        imageIds: Array.from(selectedImages),
      });
    }
  };

  // Close delete modal
  const closeDeleteModal = () => {
    setDeleteModal({ isOpen: false, imageIds: [] });
  };

  // Confirm delete
  const confirmDelete = async () => {
    setIsDeleting(true);
    try {
      const res = await removeImages(currentAlbumId, deleteModal.imageIds);
      if (res) {
        setSelectedImages(new Set());
        setLastSelectedIndex(null); // **RESET LAST SELECTED**
        closeDeleteModal();
      }
    } catch (error) {
      console.error("Error deleting images:", error);
    } finally {
      setIsDeleting(false);
    }
  };

  const { page, limit, totalItems, totalPages } = pagination || {};

  return (
    <div className="relative">
      {/* Compact Header - Only pagination, no selection controls */}
      {uploadedImages.length > 0 && (
        <div className="bg-white rounded-lg shadow-sm border border-secondary px-3 py-2 mb-2">
          <div className="flex items-center justify-between">
            <span className="text-sm text-primary font-medium">
              {totalItems || uploadedImages.length} images
              {/* **SHOW SELECTION HINT** */}
              {!isSelectionMode && (
                <span className="text-xs text-primary/50 ml-2">
                  (Hold Shift to select multiple)
                </span>
              )}
            </span>

            {/* Pagination Controls - Only show when needed */}
            {pagination && totalPages > 1 && (
              <div className="flex items-center gap-3">
                <div className="text-xs text-primary/70">
                  Showing {(page - 1) * limit + 1} -{" "}
                  {Math.min(page * limit, totalItems)} of {totalItems}
                </div>

                <div className="flex items-center gap-2">
                  {/* Items per page */}
                  <div className="flex items-center gap-1">
                    <span className="text-xs text-primary/70">Show:</span>
                    <select
                      value={limit}
                      onChange={(e) => onLimitChange(e.target.value)}
                      className="text-xs border border-secondary rounded px-1 py-0.5 text-primary focus:border-primary focus:outline-none"
                    >
                      {[100, 300, 500, 1000].map((option) => (
                        <option key={option} value={option}>
                          {option}
                        </option>
                      ))}
                    </select>
                  </div>

                  {/* Pagination buttons */}
                  <div className="flex items-center gap-1">
                    <button
                      onClick={() => onPageChange(page - 1)}
                      disabled={page === 1}
                      className={`p-1 rounded ${
                        page === 1
                          ? "text-primary/40 cursor-not-allowed"
                          : "text-primary hover:bg-secondary/30"
                      }`}
                    >
                      <FiChevronLeft size={14} />
                    </button>

                    {Array.from({ length: Math.min(5, totalPages) }, (_, i) => {
                      let pageNum;
                      if (totalPages <= 5) {
                        pageNum = i + 1;
                      } else if (page <= 3) {
                        pageNum = i + 1;
                      } else if (page >= totalPages - 2) {
                        pageNum = totalPages - 4 + i;
                      } else {
                        pageNum = page - 2 + i;
                      }

                      return (
                        <button
                          key={pageNum}
                          onClick={() => onPageChange(pageNum)}
                          className={`w-6 h-6 rounded text-xs ${
                            page === pageNum
                              ? "bg-primary text-white"
                              : "text-primary hover:bg-secondary/30"
                          }`}
                        >
                          {pageNum}
                        </button>
                      );
                    })}

                    {totalPages > 5 && page < totalPages - 2 && (
                      <span className="text-xs text-primary/40 px-1">...</span>
                    )}

                    {totalPages > 5 && page < totalPages - 2 && (
                      <button
                        onClick={() => onPageChange(totalPages)}
                        className={`w-6 h-6 rounded text-xs ${
                          page === totalPages
                            ? "bg-primary text-white"
                            : "text-primary hover:bg-secondary/30"
                        }`}
                      >
                        {totalPages}
                      </button>
                    )}

                    <button
                      onClick={() => onPageChange(page + 1)}
                      disabled={page === totalPages}
                      className={`p-1 rounded ${
                        page === totalPages
                          ? "text-primary/40 cursor-not-allowed"
                          : "text-primary hover:bg-secondary/30"
                      }`}
                    >
                      <FiChevronRight size={14} />
                    </button>
                  </div>
                </div>
              </div>
            )}
          </div>
        </div>
      )}

      {/* Image Grid */}
      <div
        className={`grid gap-2 xl:gap-3 transition-all duration-300 ${
          isSelectionMode
            ? "grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 xl:grid-cols-7"
            : "grid-cols-3 md:grid-cols-4 lg:grid-cols-6"
        }`}
      >
        {uploadedImages.length ? (
          uploadedImages.map((image, idx) => (
            <div
              key={idx}
              className={`group relative bg-secondary/20 rounded-lg overflow-hidden aspect-square shadow-sm hover:shadow-md transition-all duration-200 ${
                selectedImages.has(image._id) ? "ring-2 ring-primary" : ""
              }`}
            >
              <Image
                src={`${config.IMG_BASE_URL}${image?.thumbnail?.path}`}
                alt={image.originalName || "photo"}
                className="w-full h-full object-cover"
                loading="lazy"
                width={500}
                height={500}
                quality={75}
                placeholder="blur"
                unoptimized
                blurDataURL="data:image/jpeg;base64,/9j/4AAQSkZJRgABAQAAAQABAAD/2wBDAAYEBQYFBAYGBQYHBwYIChAKCgkJChQODwwQFxQYGBcUFhYaHSUfGhsjHBYWICwgIyYnKSopGR8tMC0oMCUoKSj/2wBDAQcHBwoIChMKChMoGhYaKCgoKCgoKCgoKCgoKCgoKCgoKCgoKCgoKCgoKCgoKCgoKCgoKCgoKCgoKCgoKCgoKCj/wAARCAABAAEDASIAAhEBAxEB/8QAFQABAQAAAAAAAAAAAAAAAAAAAAv/xAAhEAACAQMDBQAAAAAAAAAAAAABAgMABAUGIWGRkqGx0f/EABUBAQEAAAAAAAAAAAAAAAAAAAMF/8QAGhEAAgIDAAAAAAAAAAAAAAAAAAECEgMRkf/aAAwDAQACEQMRAD8AltocmV2ZXJzaW9uOiAyLjg1LjAKY29tcGlsZXI6IGdjYyAoRGViaWFuIDEwLjItMCkgMTAuMi4wCmxhbmdfdmVyc2lvbjogQ09QRSAoR0NDKSAxMC4yLjAKbW9kZWw6IGFybTY0CWFyY2g6IGFy bQY="
              />

              {/* Selection Checkbox - Always visible when in selection mode */}
              {isSelectionMode && (
                <div className="absolute top-2 left-2 z-10">
                  <button
                    onClick={(e) => toggleImageSelection(image._id, idx, e)} // **PASS EVENT & INDEX**
                    className={`w-6 h-6 rounded-full border-2 flex items-center justify-center transition-all duration-200 shadow-lg ${
                      selectedImages.has(image._id)
                        ? "bg-primary border-primary text-white scale-110"
                        : "bg-white/95 border-secondary hover:border-primary backdrop-blur-sm"
                    }`}
                  >
                    {selectedImages.has(image._id) && <FiCheck size={14} />}
                  </button>
                </div>
              )}

              {/* Hover Actions */}
              {!isSelectionMode && (
                <div className="absolute inset-0 bg-black/0 group-hover:bg-primary/40 transition-all duration-200 flex items-center justify-center opacity-0 group-hover:opacity-100">
                  <div className="flex gap-2">
                    {/* Select Button on Hover */}
                    <button
                      onClick={(e) => toggleImageSelection(image._id, idx, e)} // **PASS EVENT & INDEX**
                      className="bg-primary text-white p-2 rounded-full hover:bg-primary/80 transition-all duration-200 transform hover:scale-110"
                      aria-label="Select image"
                      title="Select image (Hold Shift for range selection)"
                    >
                      <FiCheck size={16} />
                    </button>

                    <button
                      onClick={(e) => {
                        e.stopPropagation();
                        setDeleteModal({
                          isOpen: true,
                          imageIds: [image._id],
                        });
                      }}
                      className="bg-red-500 text-white p-2 rounded-full hover:bg-red-600 transition-all duration-200 transform hover:scale-110"
                      aria-label="Remove image"
                      title="Remove image"
                    >
                      <FiTrash2 size={16} />
                    </button>

                    <a
                      target="_blank"
                      href={`${config.IMG_BASE_URL}${image?.file?.path}`}
                      download={image.originalName}
                      className="bg-secondary text-primary p-2 rounded-full hover:bg-secondary/80 transition-all duration-200 transform hover:scale-110"
                      aria-label="Download image"
                      title="Download image"
                    >
                      <FiEye size={16} />
                    </a>
                  </div>
                </div>
              )}

              {/* **CLICKABLE AREA FOR SHIFT+CLICK** */}
              {isSelectionMode && (
                <div
                  className="absolute inset-0 cursor-pointer"
                  onClick={(e) => toggleImageSelection(image._id, idx, e)} // **ENTIRE IMAGE CLICKABLE**
                />
              )}

              {/* Image Info */}
              <div
                className={`absolute bottom-0 left-0 right-0 bg-gradient-to-t from-primary/80 to-transparent px-2 py-1.5 text-white text-[10px] sm:text-xs truncate transition-all duration-200 ${
                  isSelectionMode
                    ? "opacity-100"
                    : "opacity-0 group-hover:opacity-100"
                }`}
              >
                <p className="truncate">{image?.originalName}</p>
                <p className="text-[8px] sm:text-[10px] text-secondary/80 mt-0.5">
                  {image?.file?.sizeInMB} MB
                </p>
              </div>

              {/* Selection Overlay */}
              {isSelectionMode && selectedImages.has(image._id) && (
                <div className="absolute inset-0 bg-primary/20 rounded-lg"></div>
              )}
            </div>
          ))
        ) : (
          <div className="col-span-full text-center bg-white rounded-lg shadow-sm border border-secondary p-6 sm:p-8 md:p-12">
            <div className="max-w-sm mx-auto">
              <FiImage
                size={32}
                className="mx-auto mb-2 sm:mb-3 text-secondary"
              />
              <h3 className="text-primary font-medium mb-1 text-sm sm:text-base">
                No images uploaded yet
              </h3>
              <p className="text-xs sm:text-sm text-primary/70">
                Select files or drag and drop to add images to this album
              </p>
            </div>
          </div>
        )}
      </div>

      {/* Sticky Bottom Selection Bar - Only shows when images are selected */}
      {isSelectionMode && (
        <div className="fixed bottom-4 left-4 right-4 bg-white rounded-xl shadow-2xl border border-secondary p-4 z-50 max-w-4xl mx-auto">
          <div className="flex items-center justify-between">
            <div className="flex flex-wrap items-center gap-3">
              <button
                onClick={deselectAllImages}
                className="p-2 hover:bg-secondary/30 rounded-full transition-colors"
                title="Cancel selection"
              >
                <FiX size={20} className="text-primary" />
              </button>
              <span className="font-medium text-primary">
                {selectedImages.size} selected
              </span>
              {/* **SHIFT+CLICK HINT** */}
              <span className="text-xs text-primary/50">
                Hold Shift to select range
              </span>
            </div>

            <div className="flex flex-wrap items-center gap-2">
              <button
                onClick={selectAllImages}
                className="flex items-center gap-1 px-3 py-1.5 text-sm text-primary hover:bg-secondary/30 rounded-lg transition-colors"
              >
                <MdSelectAll size={16} />
                Select All
              </button>
              <button
                onClick={deselectAllImages}
                className="flex items-center gap-1 px-3 py-1.5 text-sm text-primary/70 hover:bg-secondary/20 rounded-lg transition-colors"
              >
                <MdCheckBoxOutlineBlank size={16} />
                Deselect All
              </button>

              <button
                onClick={handleMultiDelete}
                className="flex items-center gap-1 px-4 py-2 bg-red-500 text-white rounded-lg hover:bg-red-600 transition-colors font-medium ml-2"
              >
                <FiTrash2 size={16} />
                Delete ({selectedImages.size})
              </button>
            </div>
          </div>
        </div>
      )}

      {/* Delete Modal */}
      {deleteModal.isOpen && (
        <DeleteImage
          isOpen={deleteModal.isOpen}
          onClose={closeDeleteModal}
          onConfirm={confirmDelete}
          count={deleteModal.imageIds.length}
          isLoading={isDeleting}
        />
      )}
    </div>
  );
};

export default ImageGallery;
