function LangCycleSub() {
  const [idx, setIdx] = useState(0);

  useEffect(() => {
    const t = setInterval(
      () => setIdx((i) => (i + 1) % LANG_SUBS.length),
      3500,
    );
    return () => clearInterval(t);
  }, []);

  const current = LANG_SUBS[idx];

  return (
    <div className="flex flex-col items-center gap-3 mb-8 max-w-md mx-auto">
      <AnimatePresence mode="wait">
        <motion.div
          key={`badge-${idx}`}
          initial={{ opacity: 0, scale: 0.8, y: 5 }}
          animate={{ opacity: 1, scale: 1, y: 0 }}
          exit={{ opacity: 0, scale: 0.8, y: -5 }}
          className="flex items-center gap-2 px-4 py-1.5 rounded-full bg-white/5 border border-white/10 backdrop-blur-md"
        >
          <Globe className="w-3 h-3" style={{ color: current.color }} />
          <span className="text-[10px] font-bold tracking-wider uppercase text-white/80">
            {current.label}
          </span>
        </motion.div>
      </AnimatePresence>

      <div className="relative h-12 w-full flex items-center justify-center overflow-hidden">
        <AnimatePresence mode="wait">
          <motion.p
            key={`sub-${idx}`}
            initial={{ opacity: 0, y: 15, filter: "blur(8px)" }}
            animate={{ opacity: 1, y: 0, filter: "blur(0px)" }}
            exit={{ opacity: 0, y: -15, filter: "blur(8px)" }}
            transition={{ duration: 0.5, ease: EASE }}
            className="absolute text-base sm:text-lg text-white/60 font-medium leading-relaxed text-center px-4"
          >
            {current.text}
          </motion.p>
        </AnimatePresence>
      </div>
    </div>
  );
}

function SearchBox({ query, setQuery, onSubmit, onChip }) {
  const [focused, setFocused] = useState(false);
  const [idx, setIdx] = useState(0);
  const [typed, setTyped] = useState("");
  const [del, setDel] = useState(false);

  useEffect(() => {
    const cur = PHRASES[idx];
    let t;
    if (!del && typed.length < cur.length)
      t = setTimeout(() => setTyped(cur.slice(0, typed.length + 1)), 60);
    else if (!del) t = setTimeout(() => setDel(true), 2500);
    else if (del && typed.length > 0)
      t = setTimeout(() => setTyped(typed.slice(0, -1)), 30);
    else {
      setDel(false);
      setIdx((i) => (i + 1) % PHRASES.length);
    }
    return () => clearTimeout(t);
  }, [typed, del, idx]);

  return (
    <div className="w-full max-w-2xl mx-auto">
      <form onSubmit={onSubmit} className="relative group">
        <div
          className={`absolute -inset-1 rounded-[2rem] blur-xl transition-all duration-500 opacity-20 group-hover:opacity-40 ${focused ? "opacity-60 scale-105" : ""}`}
          style={{ background: `linear-gradient(135deg, ${BL}, ${GR})` }}
        />
        <div
          className={`relative flex items-center bg-white/5 backdrop-blur-2xl border transition-all duration-300 rounded-[1.5rem] overflow-hidden ${focused ? "border-white/30 bg-white/10 ring-4 ring-white/5" : "border-white/10"}`}
        >
          <div className="pl-6">
            <Search
              className={`w-5 h-5 transition-colors ${focused ? "text-white" : "text-white/40"}`}
            />
          </div>
          <input
            type="text"
            value={query}
            onChange={(e) => setQuery(e.target.value)}
            onFocus={() => setFocused(true)}
            onBlur={() => setFocused(false)}
            placeholder={
              !focused && !query ? typed : "Search or paste product link…"
            }
            className="flex-1 py-5 px-4 text-base text-white placeholder:text-white/30 bg-transparent focus:outline-none font-medium"
          />
          <motion.button
            type="submit"
            whileHover={{ scale: 1.05 }}
            whileTap={{ scale: 0.95 }}
            className="mr-3 px-8 py-3 rounded-xl text-sm font-bold text-white bg-[#0167ff] shadow-xl shadow-blue-500/20 hover:shadow-blue-500/40 transition-shadow"
          >
            <span className="flex items-center gap-2">
              <Zap className="w-4 h-4 fill-current" />
              Search
            </span>
          </motion.button>
        </div>
      </form>

      <div className="flex flex-wrap items-center justify-center gap-3 mt-6">
        <span className="text-[10px] font-bold text-white/30 uppercase tracking-[0.2em] flex items-center gap-2">
          <Flame className="w-3.5 h-3.5 text-orange-500" /> Trending:
        </span>
        {TRENDING.map((t, i) => (
          <motion.button
            key={i}
            onClick={() => onChip(t)}
            whileHover={{
              y: -2,
              color: "#fff",
              borderColor: "rgba(255,255,255,0.3)",
            }}
            className="px-3 py-1 rounded-full text-[10px] font-bold text-white/50 border border-white/10 bg-white/5 transition-all"
          >
            {t}
          </motion.button>
        ))}
      </div>
    </div>
  );
}
