"use client";
import { trackProductClick } from "@/src/services/profileService";
import { motion } from "framer-motion";
import {
  ExternalLink,
  TrendingDown,
  Zap,
  ShieldCheck,
  Star,
} from "lucide-react";

// ── Colour tokens ─────────────────────────────────────────────────────────────
export const P_DARK = "#0167ff";
export const P_GREEN = "#34c759";
export const P_GREEN_LIGHT = "#e6f9ec";

// ── Helpers ───────────────────────────────────────────────────────────────────
const fmt = (n) => `₹${Number(n).toLocaleString("en-IN")}`;
const pct = (n) => `${Math.round(n)}%`;
const score = (n) => `${Math.round(Number(n) * 100)}%`;

// ─────────────────────────────────────────────────────────────────────────────
// SourceProductCard
export const SourceProductCard = ({ product }) => {
  if (!product) return null;

  const handleTrackClick = async () => {
    try {
      await trackProductClick(
        product.platform,
        product.url,
        product.price,
        product.originalPrice || product.price,
        0,
      );
    } catch (e) {}
  };

  const hasImage = !!product.image;

  return (
    <motion.div
      initial={{ opacity: 0, y: 10 }}
      animate={{ opacity: 1, y: 0 }}
      className="bg-white rounded-3xl border border-blue-50 shadow-[0_8px_30px_rgb(0,0,0,0.04)] overflow-hidden lg:sticky lg:top-24"
    >
      {/* ── Mobile strip ── */}
      <div className="flex items-start gap-3.5 p-3.5 lg:hidden">
        <div className="w-[80px] h-[80px] flex-shrink-0 bg-gray-50 rounded-2xl overflow-hidden border border-gray-100 flex items-center justify-center p-2">
          {hasImage ? (
            <img
              src={product.image}
              alt={product.title}
              className="w-full h-full object-contain mix-blend-multiply"
            />
          ) : (
            <span className="text-[8px] font-black text-gray-300 uppercase tracking-widest text-center leading-tight">
              No Photo
            </span>
          )}
        </div>

        <div className="flex-1 min-w-0">
          <div className="flex items-center gap-1.5 mb-1.5">
            <span className="px-1.5 py-0.5 rounded-lg text-[7px] font-black uppercase tracking-[0.1em] bg-[#0167ff] text-white">
              {product.platform}
            </span>
            {product.brand && (
              <span className="text-[8px] font-black text-gray-300 uppercase tracking-widest truncate">
                {product.brand}
              </span>
            )}
          </div>

          <p className="text-[11px] font-black text-gray-900 leading-snug line-clamp-2 mb-2 tracking-tight">
            {product.title}
          </p>

          <div className="flex items-baseline gap-2 mb-2.5">
            <span className="text-sm font-black text-[#0167ff] tabular-nums">
              {fmt(product.price)}
            </span>
            {product.originalPrice && (
              <span className="text-[9px] text-gray-300 line-through tabular-nums">
                {fmt(product.originalPrice)}
              </span>
            )}
          </div>

          <a
            href={product.url}
            target="_blank"
            rel="noopener noreferrer"
            onClick={handleTrackClick}
            className="inline-flex items-center gap-1.5 px-3 py-1.5 rounded-xl font-black text-[9px] text-white active:scale-95 transition-all shadow-lg shadow-[#0167ff]/10 uppercase tracking-widest"
            style={{ background: P_DARK }}
          >
            Visit Shop
            <ExternalLink size={10} />
          </a>
        </div>
      </div>

      {/* ── Desktop vertical card ── */}
      <div className="hidden lg:block p-6">
        <div className="relative aspect-square bg-gray-50 rounded-3xl overflow-hidden mb-5 group p-6">
          {hasImage ? (
            <img
              src={product.image}
              alt={product.title}
              className="w-full h-full object-contain mix-blend-multiply transition-transform duration-700 group-hover:scale-105"
            />
          ) : (
            <div className="w-full h-full flex items-center justify-center">
              <span className="text-[10px] font-black text-gray-300 uppercase tracking-widest">
                Product Image Not Available
              </span>
            </div>
          )}
        </div>

        <div className="space-y-4">
          <div className="flex items-center gap-2">
            <span className="px-2 py-0.5 rounded-lg text-[8px] font-black uppercase tracking-widest bg-[#0167ff] text-white">
              {product.platform}
            </span>
            {product.brand && (
              <span className="text-[9px] font-black text-gray-300 uppercase tracking-widest">
                {product.brand}
              </span>
            )}
          </div>

          <h2 className="text-xs font-black text-gray-900 leading-relaxed tracking-tight line-clamp-3">
            {product.title}
          </h2>

          <div className="flex items-baseline gap-2 pt-1">
            <span className="text-2xl font-black text-[#0167ff] tabular-nums">
              {fmt(product.price)}
            </span>
            {product.originalPrice && (
              <span className="text-xs text-gray-300 line-through tabular-nums">
                {fmt(product.originalPrice)}
              </span>
            )}
          </div>

          <a
            href={product.url}
            target="_blank"
            rel="noopener noreferrer"
            onClick={handleTrackClick}
            className="w-full inline-flex items-center justify-center gap-2 px-5 py-3.5 rounded-2xl font-black text-[10px] text-white active:scale-95 transition-all shadow-xl shadow-[#0167ff]/20 uppercase tracking-[0.15em]"
            style={{ background: P_DARK }}
          >
            Go to {product.platform}
            <ExternalLink size={14} />
          </a>
        </div>
      </div>
    </motion.div>
  );
};

// ComparisonCard
export const ComparisonCard = ({
  item,
  isBestPrice,
  sourcePrice,
  totalSaving,
}) => {
  const { platform, match } = item;
  if (!match) return null;

  const cheaper = sourcePrice > match.price;
  const priceDiff = Math.abs(sourcePrice - match.price);
  const matchPct = match.matchScore ? score(match.matchScore) : null;

  const handleTrackClick = async () => {
    try {
      await trackProductClick(
        platform,
        match.url,
        match.price,
        sourcePrice,
        totalSaving || 0,
      );
    } catch (e) {}
  };

  return (
    <motion.div
      initial={{ opacity: 0, y: 8 }}
      animate={{ opacity: 1, y: 0 }}
      className={`relative rounded-3xl bg-white border transition-all duration-300
        ${
          isBestPrice
            ? "border-[#34c759]/30 shadow-[0_10px_30px_rgb(52,199,89,0.1)] ring-1 ring-[#34c759]/5"
            : "border-gray-100 shadow-[0_4px_20px_rgb(0,0,0,0.02)]"
        }`}
    >
      {isBestPrice && (
        <div
          className="flex items-center justify-between px-3.5 py-1.5"
          style={{ background: P_GREEN }}
        >
          <div className="flex items-center gap-1.5">
            <Zap size={10} className="text-white fill-white" />
            <span className="text-[8px] font-black text-white uppercase tracking-[0.2em]">
              Best Match Found
            </span>
          </div>
          {totalSaving > 0 && (
            <span className="text-[8px] font-black text-white bg-white/20 px-1.5 py-0.5 rounded">
              SAVE {fmt(totalSaving)}
            </span>
          )}
        </div>
      )}

      <div className="flex items-start gap-4 p-4">
        <div className={`flex-shrink-0 bg-gray-50 rounded-2xl overflow-hidden border p-2 ${isBestPrice ? "border-[#34c759]/20" : "border-gray-50"}`} style={{ width: 64, height: 64 }}>
          {match.image ? (
            <img
              src={match.image}
              alt={match.title}
              className="w-full h-full object-contain mix-blend-multiply"
            />
          ) : (
            <div className="w-full h-full flex items-center justify-center">
              <span className="text-[7px] font-black text-gray-300 uppercase text-center leading-tight">No Photo</span>
            </div>
          )}
        </div>

        <div className="flex-1 min-w-0">
          <div className="flex items-center gap-2 mb-1 flex-wrap">
            <span className="text-[9px] font-black uppercase tracking-widest text-[#0167ff]">
              {platform}
            </span>
            {matchPct && (
              <span className="inline-flex items-center gap-0.5 px-1.5 py-0.5 rounded bg-gray-50 border border-gray-100 text-[7px] font-black text-gray-400 uppercase tracking-tighter">
                <ShieldCheck size={8} className="text-[#0167ff]" />
                {matchPct} Match
              </span>
            )}
          </div>

          <p className="text-[10px] font-bold text-gray-900 line-clamp-2 mb-2 leading-snug">
            {match.title}
          </p>

          <div className="flex items-center gap-2 flex-wrap">
            <span className={`text-sm font-black tabular-nums ${isBestPrice ? "text-[#34c759]" : "text-gray-900"}`}>
              {fmt(match.price)}
            </span>
            {sourcePrice && (
              <span className="text-[9px] text-gray-300 tabular-nums">
                vs <span className="line-through">{fmt(sourcePrice)}</span>
              </span>
            )}
            {cheaper ? (
              <div className="flex items-center gap-0.5 text-[8px] font-black text-[#34c759] bg-[#34c759]/5 px-1.5 py-0.5 rounded-lg border border-[#34c759]/10">
                <TrendingDown size={8} />
                {fmt(priceDiff)} LESS
              </div>
            ) : priceDiff > 0 ? (
              <div className="text-[8px] font-black text-orange-400 bg-orange-50 px-1.5 py-0.5 rounded-lg border border-orange-100">
                +{fmt(priceDiff)}
              </div>
            ) : null}
          </div>
        </div>

        <a
          href={match.url}
          target="_blank"
          rel="noopener noreferrer"
          onClick={handleTrackClick}
          className="flex-shrink-0 self-center w-8 h-8 rounded-full flex items-center justify-center text-white shadow-lg active:scale-90 transition-all"
          style={{ background: isBestPrice ? P_GREEN : P_DARK }}
        >
          <ExternalLink size={14} />
        </a>
      </div>
    </motion.div>
  );
};
