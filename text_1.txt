export const ShoppingRadar = ({ stageIdx = 0 }) => {
  const SIZE = 200;
  const CX = SIZE / 2;
  const CY = SIZE / 2;
  const R = 64;
  const INNER_R = 13;
  const BLIP_R = R - 10;
  const LABEL_R = R + 22;

  const [angle, setAngle] = useState(0);
  const rafRef = useRef(null);
  const startRef = useRef(Date.now());
  const PERIOD = 3200;

  useEffect(() => {
    const tick = () => {
      setAngle((((Date.now() - startRef.current) % PERIOD) / PERIOD) * 360);
      rafRef.current = requestAnimationFrame(tick);
    };
    rafRef.current = requestAnimationFrame(tick);
    return () => cancelAnimationFrame(rafRef.current);
  }, []);

  const pt = (deg, r = R) => ({
    x: CX + r * Math.cos(((deg - 90) * Math.PI) / 180),
    y: CY + r * Math.sin(((deg - 90) * Math.PI) / 180),
  });

  const getAnchor = (lx) => {
    if (lx < CX - 10) return "end";
    if (lx > CX + 10) return "start";
    return "middle";
  };

  const donutSlice = (startDeg, endDeg) => {
    const os = pt(startDeg, R);
    const oe = pt(endDeg, R);
    const is = pt(startDeg, INNER_R);
    const ie = pt(endDeg, INNER_R);
    const lg = endDeg - startDeg > 180 ? 1 : 0;
    return [
      `M ${is.x.toFixed(2)} ${is.y.toFixed(2)}`,
      `L ${os.x.toFixed(2)} ${os.y.toFixed(2)}`,
      `A ${R} ${R} 0 ${lg} 1 ${oe.x.toFixed(2)} ${oe.y.toFixed(2)}`,
      `L ${ie.x.toFixed(2)} ${ie.y.toFixed(2)}`,
      `A ${INNER_R} ${INNER_R} 0 ${lg} 0 ${is.x.toFixed(2)} ${is.y.toFixed(2)}`,
      "Z",
    ].join(" ");
  };

  const LAYERS = [
    { deg: 15, opacity: 0.3 },
    { deg: 32, opacity: 0.17 },
    { deg: 52, opacity: 0.09 },
    { deg: 72, opacity: 0.04 },
  ];

  const lineOuter = pt(0, R);
  const lineInner = pt(0, INNER_R);

  return (
    <svg
      width={SIZE}
      height={SIZE}
      viewBox={`0 0 ${SIZE} ${SIZE}`}
      style={{ overflow: "visible" }}
      aria-hidden
    >
      <defs>
        <clipPath id="radarBound">
          <circle cx={CX} cy={CY} r={R} />
        </clipPath>
        <radialGradient id="cglow2" cx="50%" cy="50%" r="50%">
          <stop offset="0%" stopColor={BL} stopOpacity="0.25" />
          <stop offset="100%" stopColor={BL} stopOpacity="0" />
        </radialGradient>
      </defs>
      {[20, 40, 64].map((r) => (
        <circle
          key={r}
          cx={CX}
          cy={CY}
          r={r}
          fill="none"
          stroke={BL}
          strokeWidth="0.75"
          opacity="0.14"
        />
      ))}
      <line
        x1={CX - R - 6}
        y1={CY}
        x2={CX + R + 6}
        y2={CY}
        stroke={BL}
        strokeWidth="0.5"
        opacity="0.10"
      />
      <line
        x1={CX}
        y1={CY - R - 6}
        x2={CX}
        y2={CY + R + 6}
        stroke={BL}
        strokeWidth="0.5"
        opacity="0.10"
      />
      <circle cx={CX} cy={CY} r={18} fill="url(#cglow2)" />
      <g
        transform={`rotate(${angle.toFixed(2)}, ${CX}, ${CY})`}
        clipPath="url(#radarBound)"
      >
        {[...LAYERS].reverse().map(({ deg, opacity }, i) => (
          <path key={i} d={donutSlice(0, deg)} fill={BL} opacity={opacity} />
        ))}
        <line
          x1={lineInner.x}
          y1={lineInner.y}
          x2={lineOuter.x}
          y2={lineOuter.y}
          stroke={BL}
          strokeWidth="2.2"
          strokeLinecap="round"
          opacity="0.95"
        />
        <circle
          cx={lineOuter.x}
          cy={lineOuter.y}
          r={2.8}
          fill={BL}
          opacity="0.85"
        />
      </g>
      {Object.entries(PLATFORMS).map(([name, theme], i) => {
        const revealed = i < stageIdx;
        const { x: bx, y: by } = pt(theme.angle, BLIP_R);
        const { x: lx, y: ly } = pt(theme.angle, LABEL_R);
        const anchor = getAnchor(lx);
        return (
          <g key={name}>
            <motion.g
              initial={{ scale: 0, opacity: 0 }}
              animate={
                revealed ? { scale: 1, opacity: 1 } : { scale: 0.2, opacity: 0 }
              }
              style={{ transformOrigin: `${bx}px ${by}px` }}
              transition={{ type: "spring", stiffness: 340, damping: 22 }}
            >
              {revealed && (
                <motion.circle
                  cx={bx}
                  cy={by}
                  r={5}
                  fill="none"
                  stroke={theme.color}
                  strokeWidth="1.2"
                  animate={{ r: [5, 14], opacity: [0.85, 0] }}
                  transition={{
                    duration: 1.5,
                    repeat: Infinity,
                    ease: "easeOut",
                  }}
                />
              )}
              <circle
                cx={bx}
                cy={by}
                r={5.5}
                fill={revealed ? theme.color : "#E5E7EB"}
              />
              <circle
                cx={bx}
                cy={by}
                r={2.2}
                fill="white"
                opacity={revealed ? 1 : 0}
              />
            </motion.g>
            {revealed && (
              <motion.g
                initial={{ opacity: 0 }}
                animate={{ opacity: 1 }}
                transition={{ delay: 0.3, duration: 0.3 }}
              >
                <line
                  x1={bx}
                  y1={by}
                  x2={lx}
                  y2={ly}
                  stroke={theme.color}
                  strokeWidth="0.8"
                  strokeDasharray="2 2"
                  opacity="0.4"
                />
                <text
                  x={lx}
                  y={ly}
                  textAnchor={anchor}
                  dominantBaseline="middle"
                  fontSize="9"
                  fontWeight="800"
                  fill={theme.color}
                  style={{ fontFamily: "DM Sans, system-ui, sans-serif" }}
                >
                  {name}
                </text>
              </motion.g>
            )}
          </g>
        );
      })}
      <circle cx={CX} cy={CY} r={6} fill={BL} />
      <circle cx={CX} cy={CY} r={2.5} fill="white" />
    </svg>
  );
};

      {particles.map((p, i) => (

export const useTypingText = (texts, speed = 65, pause = 1800) => {
  const [display, setDisplay] = useState(""),
    [textIdx, setTextIdx] = useState(0),
    [phase, setPhase] = useState("typing");
  useEffect(() => {
    const current = texts[textIdx];
    let timeout;
    if (phase === "typing") {
      if (display.length < current.length)
        timeout = setTimeout(
          () => setDisplay(current.slice(0, display.length + 1)),
          speed,
        );
      else timeout = setTimeout(() => setPhase("pause"), pause);
    } else if (phase === "pause")
      timeout = setTimeout(() => setPhase("deleting"), 300);
    else if (phase === "deleting") {
      if (display.length > 0)
        timeout = setTimeout(
          () => setDisplay(display.slice(0, -1)),
          speed / 2.2,
        );
      else {
        setTextIdx((i) => (i + 1) % texts.length);
        setPhase("typing");
      }
    }
    return () => clearTimeout(timeout);
  }, [display, phase, textIdx, texts, speed, pause]);
  return display;
};

export const InlineSearch = ({ currentQuery, onSearch }) => {
  const [value, setValue] = useState(currentQuery || ""),
    [langIdx, setLangIdx] = useState(0),
    [focused, setFocused] = useState(false);
  const inputRef = useRef(null),
    typingText = useTypingText(TYPING_EXAMPLES, 68, 2000);
  useEffect(() => {
    const t = setInterval(
      () => setLangIdx((i) => (i + 1) % LANG_BADGES.length),
      1800,
    );
    return () => clearInterval(t);
  }, []);
  const handleSubmit = (e) => {
    e.preventDefault();
    if (value.trim()) onSearch(value.trim());
  };
  return (
    <form onSubmit={handleSubmit} className="mt-2.5">
      <motion.div
        animate={
          focused
            ? { boxShadow: `0 0 0 3px ${BL}18` }
            : { boxShadow: "0 0 0 0 transparent" }
        }
        transition={{ duration: 0.2 }}
        className="flex items-center gap-2 px-3 py-2.5 rounded-2xl border-2 bg-gray-50 transition-colors duration-200"
        style={{ borderColor: focused ? BL : "#E5E7EB" }}
      >
        <Search className="w-4 h-4 text-gray-400 shrink-0" />
        <input
          ref={inputRef}
          type="text"
          value={value}
          onChange={(e) => setValue(e.target.value)}
          onFocus={() => setFocused(true)}
          onBlur={() => setFocused(false)}
          placeholder={
            focused
              ? "Try Hinglish or any Indian language..."
              : typingText
                ? typingText + "▌"
                : "Search products..."
          }
          className="flex-1 text-sm font-semibold text-gray-900 bg-transparent outline-none placeholder:text-gray-400 placeholder:font-normal min-w-0"
        />
        {value && (
          <button
            type="button"
            onClick={() => {
              setValue("");
              inputRef.current?.focus();
            }}
            className="p-1 rounded-lg text-gray-400 hover:bg-gray-100 transition-colors"
          >
            <X className="w-3.5 h-3.5" />
